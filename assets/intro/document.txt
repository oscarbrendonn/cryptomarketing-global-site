    const heroRadius = radii[2] * 0.7 + ringTS / 2;
    const heroFit = Math.min(1, Math.max(0.2, (heroTop - headerBottom - 48) / (2 * heroRadius)));
    const heroY = Math.min(-H * (mob ? 0.13 : 0.1), heroTop - cy - 24 - heroRadius * heroFit);
    G = { W, H, mob, T, M, R, radii, ringTS, heroFit, off, base, focus, wall, heroMark: { x: 0, y: heroY }, finalMark: centreOf($('finalPh')) };
    moved.forEach((e, i) => { e.style.transform = saved[i]; });
  }

  // Screen point and size at depth z, for the perspective camera
  const atDepth = (x, y, size, z, T) => ({ x: x * (PERS - z) / PERS, y: y * (PERS - z) / PERS, z, s: size / T * (PERS - z) / PERS });

  let p = 0, target = 0, last = performance.now(), t0 = last, mxp = 0, myp = 0, pmx = 0, pmy = 0;
  const track = $('track'), hero = $('hero'), mid = $('mid'), fin = $('final'), hint = $('hint'), bar = $('bar');
  const NC = CH.length, C0 = 0.27, CS = 0.074;

  function frame(now) {
    if (reduce) return;
    if (document.hidden || window.CMP_INTRO_VISIBLE === false) { last = now; requestAnimationFrame(frame); return; }
    if (!G.W) { requestAnimationFrame(frame); return; }
    const dt = Math.min(0.1, (now - last) / 1000); last = now;
    const span = track.offsetHeight - innerHeight;
    target = window.CMP_INTRO_CONTEXT?.mode === 'inline' ? (window.CMP_INTRO_PROGRESS || 0) : (span > 0 ? clamp((scrollY - track.offsetTop) / span) : 0);
    p = reduce ? target : p + (target - p) * (1 - Math.exp(-dt * 7));
    pmx += (mxp - pmx) * (1 - Math.exp(-dt * 4)); pmy += (myp - pmy) * (1 - Math.exp(-dt * 4));
    const { R, T, off, base, focus, wall, mob } = G;
    const time = reduce ? 0 : (now - t0) / 1000;

    const burst = ss(0.05, 0.25, p);
    const cw = ss(0.24, 0.29, p);
    const fw = ss(0.86, 0.97, p);
    const w = CH.map((c, k) => { const a = C0 + k * CS; return ss(a, a + 0.022, p) * (1 - ss(a + CS - 0.02, a + CS + 0.002, p)); });
    const spin = time * 0.05 + p * 1.6;
    const ox = lerp(base.x, off.x, cw), oy = lerp(base.y, off.y, cw);
    const hm0 = G.heroMark;
    const ringFade = (1 - cw) * (1 - fw);

    // Orbit lines
    orbs.forEach((el, r) => {
      const rr = G.radii[r] * lerp(0.7 * G.heroFit, 1, burst), ccx = lerp(hm0.x, ox, burst), ccy = lerp(hm0.y, oy, burst);
      el.style.width = el.style.height = (rr * 2).toFixed(1) + 'px';
      el.style.transform = `translate3d(${(ccx - rr).toFixed(1)}px,${(ccy - rr).toFixed(1)}px,-2px)`;
      el.style.opacity = (lerp(r >= 3 ? 0 : 0.55, 1, burst) * ringFade).toFixed(3);
    });

    tiles.forEach(tl => {
      // Orbits: small and soft around the hero mark, full size after the first scroll
      const b = clamp(burst * 1.5 - tl.d * 0.5);
      const rr = G.radii[tl.ring] * lerp(0.7 * G.heroFit, 1, b), ang = tl.a0 + tl.dir * spin;
      const ccx = lerp(hm0.x, ox, b), ccy = lerp(hm0.y, oy, b);
      let x = ccx + rr * Math.cos(ang), y = ccy + rr * Math.sin(ang), z = 0;
      let s = G.ringTS / T * lerp(0.75 * G.heroFit, 1, b), o = lerp(tl.ring >= 3 ? 0 : 0.45, 1, b) * (1 - cw);
      let on = false;
      for (let k = 0; k < NC; k++) {
        if (w[k] <= 0) continue;
        const j = tl.ch[k];
        if (j >= 0) {
          const n = chLogos[k].length, wj = ss(0, 1, clamp(w[k] * 1.5 - (j / n) * 0.5));
          const F = focus[k]; let f, vis = 1;
          if (F.mq) {
            const m = F.mq, row = j % m.rows, col = Math.floor(j / m.rows), dir = row % 2 ? -1 : 1;
            const u = (((col * m.sp + dir * time * m.sp * 0.32) % m.Lw) + m.Lw) % m.Lw;
            const sx = m.cxr + u - m.Lw / 2, ax = sx + G.W / 2, edge = m.sp * 0.7;
            vis = ss(m.x0 - 4, m.x0 + edge, ax) * (1 - ss(m.x1 - edge, m.x1 + 4, ax));
            f = { x: sx, y: m.cyr + (row - (m.rows - 1) / 2) * m.rowH - m.lh / 2, size: m.size, lw: m.lw, lf: m.lf };
          } else f = F.pos[tl.k];
          const q = atDepth(f.x, f.y, f.size, 60, T);
          x = lerp(x, q.x, wj); y = lerp(y, q.y, wj); z = lerp(z, q.z, wj); s = lerp(s, q.s, wj); o = lerp(o, vis, wj);
          if (wj > 0.5 && vis > 0.05) { on = true; if (tl.fs !== q.s || tl.lf !== f.lf) { tl.fs = q.s; tl.lf = f.lf; tl.el.style.setProperty('--fs', q.s.toFixed(3)); tl.el.style.setProperty('--lw', f.lw.toFixed(0) + 'px'); tl.el.style.setProperty('--lf', f.lf + 'px'); } }
        }
      }
      if (fw > 0) {
        // The wall fades in where it stands, in a wave from the centre, instead of flying across
        const wl = wall[tl.k], ft = ss(0, 1, clamp(fw * 1.6 - wl.ring * 0.9));
        if (ft > 0) { o = wl.hidden ? 0 : Math.max(o * (1 - ft), ft); x = lerp(x, wl.x, o > 0.02 && ft < 1 ? 1 : ft); y = lerp(y, wl.y, o > 0.02 && ft < 1 ? 1 : ft); z = lerp(-140, 0, ft); s = wl.size / T * lerp(0.72, 1, ft); }
      }
      tl.el.style.transform = `translate3d(${x.toFixed(1)}px,${y.toFixed(1)}px,${z.toFixed(1)}px) scale(${s.toFixed(3)})`;
      tl.el.style.opacity = o.toFixed(3);
      if (on !== tl.hi) { tl.hi = on; tl.el.classList.toggle('hi', on); }
    });

    // Brand mark: hero centre, then heart of the sphere, then above the closing line
    const hm = G.heroMark, fm = G.finalMark;
    let mx = lerp(hm.x, ox, burst), my = lerp(hm.y, oy, burst), ms = lerp(1, 0.72, burst), mo = 1 - cw;
    mx = lerp(mx, fm.x, fw); my = lerp(my, fm.y, fw); ms = lerp(ms, 1, fw); mo = lerp(mo, 1, fw);
    mk.style.transform = `translate(${mx.toFixed(1)}px,${my.toFixed(1)}px) scale(${ms.toFixed(3)})`;
    mk.style.opacity = mo.toFixed(3);

    // Copy
    const ho = 1 - ss(0.015, 0.07, p);
    hero.style.opacity = ho; hero.style.transform = `translateY(${(-(1 - ho) * 30).toFixed(1)}px)`;
    hero.style.visibility = ho < 0.01 ? 'hidden' : '';
    const mo2 = ss(0.15, 0.19, p) * (1 - ss(0.235, 0.26, p));
    mid.style.opacity = mo2; mid.style.transform = `translateY(${((1 - mo2) * 24).toFixed(1)}px)`;
    mid.style.visibility = mo2 < 0.01 ? 'hidden' : '';
    chapEls.forEach((el, k) => {
      el.style.opacity = w[k].toFixed(3);
      el.style.transform = (mob ? '' : 'translateY(-50%) ') + `translateY(${((1 - w[k]) * 26).toFixed(1)}px)`;
      el.style.visibility = w[k] < 0.01 ? 'hidden' : '';
    });
    const fo = ss(0.93, 0.985, p);
    fin.style.opacity = fo; fin.style.transform = `translateY(${((1 - fo) * 24).toFixed(1)}px)`;
    fin.style.visibility = fo < 0.01 ? 'hidden' : '';
    hint.style.opacity = 1 - ss(0, 0.025, p);
    bar.style.transform = `scaleX(${p.toFixed(4)})`;
    world.style.transform = `rotateY(${(pmx * 5).toFixed(2)}deg) rotateX(${(-pmy * 4).toFixed(2)}deg)`;
    requestAnimationFrame(frame);
  }

  if (!reduce) { measure(); requestAnimationFrame(frame); }
  let rz;
  function remeasure() { if (!reduce) { clearTimeout(rz); rz = setTimeout(measure, 0); } }
  addEventListener('resize', remeasure);
  document.addEventListener('cmp:languagechange', remeasure);
  document.addEventListener('cmp:themechange', remeasure);
  matchMedia('(prefers-reduced-motion: reduce)').addEventListener('change', () => location.reload());
  if (!reduce && matchMedia('(hover: hover)').matches) {
    addEventListener('pointermove', e => { mxp = e.clientX / innerWidth - 0.5; myp = e.clientY / innerHeight - 0.5; }, { passive: true });
  }
})();
</script>
